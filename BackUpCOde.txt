#![cfg_attr(not(debug_assertions), windows_subsystem = "windows")]

use gpui::{
    div, prelude::*, px, rgb, size, App, Bounds, Context, FocusHandle, Focusable, IntoElement,
    Render, Window, WindowBounds, WindowOptions,
};
use gpui_platform::application;
use std::fs;
use std::path::PathBuf;

struct TodoApp {
    todos: Vec<String>,
    input: String,
    focus_handle: FocusHandle,
    next_task_number: usize,
    editing_index: Option<usize>,
    edit_input: String,
}

impl Focusable for TodoApp {
    fn focus_handle(&self, _cx: &App) -> FocusHandle {
        self.focus_handle.clone()
    }
}

impl TodoApp {
    fn get_storage_path() -> PathBuf {
        let mut path = if let Ok(appdata) = std::env::var("APPDATA") {
            PathBuf::from(appdata)
        } else if let Ok(home) = std::env::var("HOME") {
            PathBuf::from(home)
        } else if let Ok(userprofile) = std::env::var("USERPROFILE") {
            PathBuf::from(userprofile)
        } else {
            std::env::current_dir().unwrap_or_default()
        };
        path.push("todo-app-gpui");
        path.push("todos.json");
        path
    }

    fn load_todos() -> Option<Vec<String>> {
        let path = Self::get_storage_path();
        if path.exists() {
            let content = fs::read_to_string(path).ok()?;
            serde_json::from_str(&content).ok()
        } else {
            None
        }
    }

    fn save_todos(&self) {
        let path = Self::get_storage_path();
        if let Some(parent) = path.parent() {
            let _ = fs::create_dir_all(parent);
        }
        if let Ok(content) = serde_json::to_string_pretty(&self.todos) {
            let _ = fs::write(path, content);
        }
    }

    fn new(cx: &mut Context<Self>) -> Self {
        let todos = Self::load_todos().unwrap_or_default();
        let next_task_number = todos.len() + 1;
        Self {
            todos,
            input: String::new(),
            focus_handle: cx.focus_handle(),
            next_task_number,
            editing_index: None,
            edit_input: String::new(),
        }
    }

    fn add_todo(&mut self, cx: &mut Context<Self>) {
        let task = self.input.trim().to_owned();

        if !task.is_empty() {
            self.todos.push(task);
            self.input.clear();
            self.next_task_number += 1;
            self.save_todos();
            cx.notify();
        }
    }

    fn handle_key(&mut self, event: &gpui::KeyDownEvent, cx: &mut Context<Self>) {
        let key = event.keystroke.key.as_str();

        if self.editing_index.is_some() {
            if key.eq_ignore_ascii_case("enter") {
                self.save_edit(cx);
            } else if key.eq_ignore_ascii_case("escape") {
                self.cancel_edit(cx);
            } else if key.eq_ignore_ascii_case("backspace") {
                self.edit_input.pop();
                cx.notify();
            } else if !event.keystroke.modifiers.control && !event.keystroke.modifiers.platform {
                if let Some(character) = event.keystroke.key_char.as_deref() {
                    self.edit_input.push_str(character);
                    cx.notify();
                }
            }
        } else {
            if key.eq_ignore_ascii_case("enter") {
                self.add_todo(cx);
            } else if key.eq_ignore_ascii_case("backspace") {
                self.input.pop();
                cx.notify();
            } else if !event.keystroke.modifiers.control && !event.keystroke.modifiers.platform {
                if let Some(character) = event.keystroke.key_char.as_deref() {
                    self.input.push_str(character);
                    cx.notify();
                }
            }
        }
    }

    fn remove_todo(&mut self, index: usize, cx: &mut Context<Self>) {
        if Some(index) == self.editing_index {
            self.editing_index = None;
            self.edit_input.clear();
        } else if let Some(edit_idx) = self.editing_index {
            if index < edit_idx {
                self.editing_index = Some(edit_idx - 1);
            }
        }
        if index < self.todos.len() {
            self.todos.remove(index);
            self.save_todos();
            cx.notify();
        }
    }

    fn clear_todos(&mut self, cx: &mut Context<Self>) {
        if !self.todos.is_empty() {
            self.todos.clear();
            self.editing_index = None;
            self.edit_input.clear();
            self.save_todos();
            cx.notify();
        }
    }

    fn start_edit(&mut self, index: usize, cx: &mut Context<Self>) {
        if index < self.todos.len() {
            self.editing_index = Some(index);
            self.edit_input = self.todos[index].clone();
            cx.notify();
        }
    }

    fn save_edit(&mut self, cx: &mut Context<Self>) {
        if let Some(index) = self.editing_index {
            let updated = self.edit_input.trim().to_owned();
            if !updated.is_empty() {
                self.todos[index] = updated;
                self.editing_index = None;
                self.edit_input.clear();
                self.save_todos();
                cx.notify();
            }
        }
    }

    fn cancel_edit(&mut self, cx: &mut Context<Self>) {
        self.editing_index = None;
        self.edit_input.clear();
        cx.notify();
    }
}

impl Render for TodoApp {
    fn render(&mut self, _window: &mut Window, cx: &mut Context<Self>) -> impl IntoElement {
        let input_text = if self.input.is_empty() {
            "Type a task and press Enter".to_owned()
        } else {
            self.input.clone()
        };

        div()
            .id("todo-app")
            .track_focus(&self.focus_handle)
            .on_key_down(cx.listener(|this, event, _, cx| {
                this.handle_key(event, cx);
            }))
            .flex()
            .flex_col()
            .size_full()
            .p(px(20.0))
            .gap(px(12.0))
            .bg(rgb(0x1e1e1e))
            .text_color(rgb(0xd4d4d4))
            .font_family("Segoe UI")
            .child(
                div()
                    .text_size(px(28.0))
                    .font_weight(gpui::FontWeight::BOLD)
                    .mb(px(8.0))
                    .text_color(rgb(0x0078d4))
                    .child("TODO Application"),
            )
            .child(
                div()
                    .id("todo-input")
                    .flex()
                    .items_center()
                    .h(px(36.0))
                    .px(px(12.0))
                    .bg(rgb(0x2d2d2d))
                    .border_1()
                    .border_color(rgb(0x3d3d3d))
                    .rounded(px(6.0))
                    .text_size(px(14.0))
                    .text_color(if self.input.is_empty() {
                        rgb(0x808080)
                    } else {
                        rgb(0xd4d4d4)
                    })
                    .on_click(cx.listener(|this, _, window, cx| {
                        this.focus_handle.focus(window, cx);
                    }))
                    .child(input_text)
                    .child(div().w(px(1.0)).h(px(18.0)).bg(rgb(0x0078d4))),
            )
            .child(
                div()
                    .flex()
                    .justify_between()
                    .items_center()
                    .pb(px(8.0))
                    .border_b_1()
                    .border_color(rgb(0x3d3d3d))
                    .text_color(rgb(0x808080))
                    .text_size(px(12.0))
                    .child("Tasks")
                    .child(format!("{} items", self.todos.len())),
            )
            .child(
                div()
                    .h(px(360.0))
                    .id("todo-list")
                    .overflow_y_scroll()
                    .flex()
                    .flex_col()
                    .gap(px(8.0))
                    .children(self.todos.iter().enumerate().map(|(index, todo)| {
                        let todo_text = todo.clone();
                        let is_editing = self.editing_index == Some(index);

                        if is_editing {
                            div()
                                .flex()
                                .items_center()
                                .justify_between()
                                .p(px(10.0))
                                .bg(rgb(0x2d2d2d))
                                .rounded(px(6.0))
                                .border_1()
                                .border_color(rgb(0x0078d4))
                                .child(
                                    div()
                                        .id(("edit-input", index))
                                        .flex()
                                        .items_center()
                                        .flex_1()
                                        .h(px(28.0))
                                        .px(px(8.0))
                                        .bg(rgb(0x1e1e1e))
                                        .border_1()
                                        .border_color(rgb(0x3d3d3d))
                                        .rounded(px(4.0))
                                        .text_size(px(14.0))
                                        .text_color(rgb(0xd4d4d4))
                                        .child(self.edit_input.clone())
                                        .child(div().w(px(1.0)).h(px(14.0)).bg(rgb(0x0078d4))),
                                )
                                .child(
                                    div()
                                        .flex()
                                        .gap(px(6.0))
                                        .ml(px(8.0))
                                        .child(
                                            div()
                                                .id(("save", index))
                                                .px(px(8.0))
                                                .py(px(4.0))
                                                .bg(rgb(0x0078d4))
                                                .rounded(px(4.0))
                                                .cursor_pointer()
                                                .hover(|style| style.bg(rgb(0x106ebe)))
                                                .text_color(rgb(0xffffff))
                                                .text_size(px(12.0))
                                                .on_click(cx.listener(move |this, _, _, cx| {
                                                    this.save_edit(cx);
                                                }))
                                                .child("Save"),
                                        )
                                        .child(
                                            div()
                                                .id(("cancel", index))
                                                .px(px(8.0))
                                                .py(px(4.0))
                                                .bg(rgb(0x3d3d3d))
                                                .rounded(px(4.0))
                                                .cursor_pointer()
                                                .hover(|style| style.bg(rgb(0x4d4d4d)))
                                                .text_color(rgb(0xd4d4d4))
                                                .text_size(px(12.0))
                                                .on_click(cx.listener(move |this, _, _, cx| {
                                                    this.cancel_edit(cx);
                                                }))
                                                .child("Cancel"),
                                        ),
                                )
                        } else {
                            div()
                                .flex()
                                .items_center()
                                .justify_between()
                                .p(px(10.0))
                                .bg(rgb(0x2d2d2d))
                                .rounded(px(6.0))
                                .hover(|style| style.bg(rgb(0x333333)))
                                .child(div().text_size(px(14.0)).child(todo_text))
                                .child(
                                    div()
                                        .flex()
                                        .gap(px(6.0))
                                        .child(
                                            div()
                                                .id(("edit", index))
                                                .px(px(8.0))
                                                .py(px(4.0))
                                                .bg(rgb(0x3d3d3d))
                                                .rounded(px(4.0))
                                                .cursor_pointer()
                                                .hover(|style| style.bg(rgb(0x4d4d4d)))
                                                .text_color(rgb(0xd4d4d4))
                                                .text_size(px(12.0))
                                                .on_click(cx.listener(move |this, _, _, cx| {
                                                    this.start_edit(index, cx);
                                                }))
                                                .child("Edit"),
                                        )
                                        .child(
                                            div()
                                                .id(("delete", index))
                                                .px(px(8.0))
                                                .py(px(4.0))
                                                .bg(rgb(0x4a1a1a))
                                                .rounded(px(4.0))
                                                .cursor_pointer()
                                                .text_color(rgb(0xff6b6b))
                                                .text_size(px(12.0))
                                                .hover(|style| style.bg(rgb(0x5a2a2a)))
                                                .on_click(cx.listener(move |this, _, _, cx| {
                                                    this.remove_todo(index, cx);
                                                }))
                                                .child("Delete"),
                                        ),
                                )
                        }
                    })),
            )
            .child(
                div()
                    .mt_auto()
                    .pt(px(12.0))
                    .border_t_1()
                    .border_color(rgb(0x3d3d3d))
                    .flex()
                    .justify_between()
                    .child(
                        div()
                            .id("add")
                            .px(px(12.0))
                            .py(px(6.0))
                            .bg(rgb(0x0078d4))
                            .rounded(px(4.0))
                            .cursor_pointer()
                            .hover(|style| style.bg(rgb(0x106ebe)))
                            .text_color(rgb(0xffffff))
                            .font_weight(gpui::FontWeight::SEMIBOLD)
                            .on_click(cx.listener(|this, _, _, cx| {
                                this.add_todo(cx);
                            }))
                            .child("Add task"),
                    )
                    .child(
                        div()
                            .id("clear-all")
                            .px(px(12.0))
                            .py(px(6.0))
                            .bg(rgb(0x4a1a1a))
                            .rounded(px(4.0))
                            .cursor_pointer()
                            .hover(|style| style.bg(rgb(0x5a2a2a)))
                            .text_color(rgb(0xff6b6b))
                            .on_click(cx.listener(|this, _, _, cx| {
                                this.clear_todos(cx);
                            }))
                            .child("Clear all"),
                    ),
            )
    }
}

fn main() {
    application().run(|cx: &mut App| {
        let bounds = Bounds::centered(None, size(px(500.0), px(600.0)), cx);

        cx.open_window(
            WindowOptions {
                window_bounds: Some(WindowBounds::Windowed(bounds)),
                ..Default::default()
            },
            |window, cx| {
                let app = cx.new(TodoApp::new);
                app.focus_handle(cx).focus(window, cx);
                app
            },
        )
        .expect("failed to open the TODO window");

        cx.activate(true);
    });
}
